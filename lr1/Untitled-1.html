using System;

namespace ArrayTask
{
    // Класс, работающий с одномерным массивом
    public class CustomArray
    {
        private double[] array;

        // Конструктор с параметрами: количество элементов массива, диапазон значений элементов массива
        public CustomArray(int count, double minValue, double maxValue)
        {
            if (count <= 0)
                throw new ArgumentException("Количество элементов должно быть положительным числом", nameof(count));

            if (minValue >= maxValue)
                throw new ArgumentException("Минимальное значение должно быть меньше максимального", nameof(minValue));

            array = new double[count];
            InitializeArray(minValue, maxValue);
        }

        // Конструктор с параметрами: количество элементов массива (диапазон по умолчанию: 3.3 до 10)
        public CustomArray(int count) : this(count, 3.3, 10.0)
        {
        }

        // Инициализация массива случайными значениями в заданном диапазоне
        private void InitializeArray(double minValue, double maxValue)
        {
            Random random = new Random();

            for (int i = 0; i < array.Length; i++)
            {
                // Генерация случайного числа в заданном диапазоне
                array[i] = minValue + random.NextDouble() * (maxValue - minValue);
            }
        }

        // Метод 1: Вычисление суммы элементов массива с четными индексами
        public double CalculateSumEvenIndices()
        {
            double sum = 0;

            for (int i = 0; i < array.Length; i += 2)
            {
                sum += array[i];
            }

            return sum;
        }

        // Метод 2: Вычисление произведения элементов массива, 
        // расположенных между максимальным и минимальным элементами
        public double CalculateProductBetweenMinMax()
        {
            if (array.Length < 3) // Нужно хотя бы 3 элемента, чтобы были элементы между
                return 0;

            // Находим индексы минимального и максимального элементов
            int minIndex = 0;
            int maxIndex = 0;

            for (int i = 1; i < array.Length; i++)
            {
                if (array[i] < array[minIndex])
                    minIndex = i;
                if (array[i] > array[maxIndex])
                    maxIndex = i;
            }

            // Определяем границы для вычисления произведения
            int startIndex = Math.Min(minIndex, maxIndex) + 1;
            int endIndex = Math.Max(minIndex, maxIndex) - 1;

            // Если между элементами нет других элементов
            if (startIndex > endIndex)
                return 0;

            // Вычисляем произведение
            double product = 1;
            for (int i = startIndex; i <= endIndex; i++)
            {
                product *= array[i];
            }

            return product;
        }

        // Свойство, доступное только для чтения: среднее арифметическое элементов массива
        public double Average
        {
            get
            {
                if (array.Length == 0)
                    return 0;

                double sum = 0;
                foreach (double element in array)
                {
                    sum += element;
                }

                return sum / array.Length;
            }
        }

        // Метод для вывода массива в строку (для демонстрации)
        public void PrintArray(string arrayName)
        {
            Console.Write($"{arrayName}: [");
            for (int i = 0; i < array.Length; i++)
            {
                Console.Write($"{array[i]:F2}");
                if (i < array.Length - 1)
                    Console.Write(", ");
            }
            Console.WriteLine("]");
        }

        // Свойство для получения длины массива
        public int Length
        {
            get { return array.Length; }
        }
    }
